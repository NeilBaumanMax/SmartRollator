                    C --> D["osKernelInitialize"]

                    subgraph Task_ForceSensor [紫色: ForceSensorTask]

                        FS["ForceSensorTask"]

                        FS1 --> FS2["接收 29 字节力数据"]

                        FS2 --> FS3["processForceData()"]

                        FS3 --> FQ[("ForceQueue 队列")]

                    end

                    G --> FS


                    %% 核心控制与算法任务 (1ms 周期)

                    subgraph Task_Control [蓝色: ControlTask 1ms]

                        CT["ControlTask"]

                        CT --> C0["CAN_Init / RPDO1_Config / Driver_Enable"]

                        C0 --> C1["CPG_Init(&my_cpg)"]

                        CT --> C2["读取 ForceQueue 最新力"]

                        C2 --> C3["新鲜度检查"]

                        C3 --> CR["Controller_Run()"]

                        

                        %% CPG 与导纳核心算法

                        CR --> CPG["CPG_Step(&my_cpg, F_x, 0.001f)"]

                        CPG --> ADM["Hopf 振荡器 + 动态阻尼 + 导纳积分"]

                        ADM --> VC["v_cmd (m/s)"]

                        VC --> RPM["线速度换算转速 (rpm)"]

                        RPM --> MQ[("MotorCmdQueue 队列")]

                    end

                    G --> CT


                    %% 电机驱动发送任务

                    subgraph Task_MotorTx [绿色: MotorTxTask]

                        MT["MotorTxTask"]

                        MT --> SYNC["基于同一个 rpm 生成左右轮同步命令"]

                        SYNC --> CAN["ZLAC_SetVelocity_rpm()"]

                        CAN --> DRV["FDCAN1 -> ZLAC8015D 驱动器"]

                        DRV --> MOTOR((左右轮毂电机))

                    end

                    G --> MT

                    MQ --> MT


                    %% 日志与视觉位姿解析任务

                        U1 --> VISION["$ARUCO 视觉位姿帧解析"]

                        U1 --> CMD["h/t/p 日志命令"]

                    end

                    G --> LOG

                    end
